#!/usr/local/bin/ruby -w
require 'net/smtp'

Net::SMTP.start('mail-kmh-relay.daumkakao.io', 25) do |smtp|
    smtp.send_message "Subject: mail test\n\ncontent",
                      'wwl1400test@daum.net',
                      ['wwl1401test@daum.net', 'wwl1402test@daum.net']
end
