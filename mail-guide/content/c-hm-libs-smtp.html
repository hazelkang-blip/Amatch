/**
 * read response
 * @param rp resource pool
 * @param rb read buffer
 * @param response response message
 * @return 0 - response code starts with 2
 */
static int read_response(respool_t *rp, readbuf_t *rb, char **response) {
    int chunk_size = 1024;
    int timeout_ms = 1000;
    char *last = NULL;
    strbuf_t *stb = rp_create_strbuf(rp, chunk_size);
    if (stb == NULL) {
        return -1;
    }
    char *readline;
    while ((readline = rb_gets(rb, &last, timeout_ms)) != NULL) {
#ifdef DEBUG_SIMPLE_SMTP_CLIENT
        fprintf(stderr, "S: %s", readline);
#endif
        stb_addstr(stb, readline);
        // check last line of response
        if (readline[3] == ' ') {
            break;
        }
    }
    *response = rp_strdup(rp, stb_getbuf(stb));
    // error
    if (response == NULL || (strncasecmp(*response, "2", 1) != 0)) {
        return -1;
    }
    return 0;
}

/**
 * do smtp command
 * @param rp
 * @param sb
 * @param rb
 * @param command
 * @param response
 * @return
 */
static int do_smtp_command(respool_t *rp, sendbuf_t *sb, readbuf_t *rb, char *command, char **response) {
//    fprintf(stderr, "%s", command); // DEBUG
    sb_write_strs(sb, command, "\n", NULL);
    sb_flush(sb);
#ifdef DEBUG_SIMPLE_SMTP_CLIENT
    fprintf(stderr, "C: %s\n", command);
#endif
    return read_response(rp, rb, response);
}

/**
 * do smtp data
 * @param rp
 * @param sb
 * @param rb
 * @param command
 * @param response
 * @return
 */
static int do_smtp_data(respool_t *rp, sendbuf_t *sb, readbuf_t *rb, char *command, char **response) {
    sb_write_strs(sb, command, "\n", NULL);
    sb_flush(sb);
#ifdef DEBUG_SIMPLE_SMTP_CLIENT
    fprintf(stderr, "C: (data...)\n");
#endif
    return read_response(rp, rb, response);
}

int sendmail_hmail(respool_t *rp, const char *from, char **recv, const char *filepath) {
    int port = 2525;
    char* host = "hmail-express.daumkakao.io";
//    char* host = "localhost";
    int timeout_ms = 1000;
    int maxchunk = 1024;
    int bufsize = 1024;
    char *response = NULL;
    int rc = 0;

    // read email file
    int chunk_size = 1024;
    strbuf_t *stb = rp_create_strbuf(rp, chunk_size);
    rc = file2strbuf(stb, filepath);
    if (rc) {
        return SMTP_ERROR_FILE;
    }
    char *message = rp_strdup(rp, stb_getbuf(stb));
//    fprintf(stderr, "%s\n", message); // debug

    // connect to smtp host
    int sd = connect_tcp(host, port, timeout_ms);
    if (sd < 0) return SMTP_ERROR_CONNECTION;

    readbuf_t *rb = NULL;
    sendbuf_t *sb = NULL;

    // create buffer for smtp
    sb = rp_create_sendbuf(rp, sd, 4096, timeout_ms);
    if (sb == NULL) {
        return -3;
    }
    rb = rp_create_readbuf(rp, sd, bufsize, maxchunk);
    if (rb == NULL) {
        return -4;
    }

    // read greeting
    rc = read_response(rp, rb, &response);
    if (rc) {
        return SMTP_ERROR_GREETING;
    }

    // EHLO
    rc = do_smtp_command(rp, sb, rb, "EHLO daum.net", &response);
    if (rc) {
        return SMTP_ERROR_EHLO;
    }

    // MAIL FROM
    char* from_command = hm_util_rp_make_string(rp, "MAIL FROM: %s", from);
//    fprintf(stderr, "%s\n", from_command); // debug
    rc = do_smtp_command(rp, sb, rb, from_command, &response);
    if (rc) {
        return SMTP_ERROR_FROM;
    }

    // RCPT TO
    int count_rcpt = 0;
    int i = 0;
    while (recv[i] != NULL) {
        if (recv[i][0] == '\0') {
            i++;
            continue;
        }
        char* recipient = recv[i++];
        do_smtp_command(rp, sb, rb, hm_util_rp_make_string(rp, "RCPT TO: %s", recipient), &response);
        // rcpt to fail
        if (response == NULL) {
            return SMTP_ERROR_RCPT;
        }
        // count when success
        if (strncasecmp(response, "2", 1) == 0) {
            count_rcpt++;
        }
        // skip for other responses
    }

    // check is there any passed rcpt
    if (count_rcpt == 0) {
        return SMTP_ERROR_RCPT_ZERO; // rcpt count is 0
    }

    // DATA
    do_smtp_command(rp, sb, rb, "DATA", &response);
    if (response == NULL || (strncasecmp(response, "3", 1) != 0)) {
        return SMTP_ERROR_DATA;
    }
    rc = do_smtp_data(rp, sb, rb, hm_util_rp_make_string(rp, "%s\n.", message), &response);
    if (rc) {
        return SMTP_ERROR_DATA_END;
    }

    // QUIT
    rc = do_smtp_command(rp, sb, rb, "QUIT", &response);
    if (rc) {
        return SMTP_ERROR_QUIT;
    }

    return 0;
}
