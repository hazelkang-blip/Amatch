perl — MIME::Lite
use MIME::Lite;

my $msg = mime_msg($header_from, $to, $subject, $contents);
$raw_msg = $msg->as_string;

sub mime_msg {
    my $from = shift;
    my $to = shift;
    my $subject = shift;
    my $contents = shift;

    my $msg = MIME::Lite->new(
        From    => $header_from,
        To      => $to,
        Subject => $subject,
        Type    => 'multipart/mixed'
    );

    $msg->attach(
        Type     => 'TEXT',
        Data     => $contents
    );

    return $msg;
}
